FROM products

UPDATE products
SET product_code = CONCAT(product_code, '_', id)
WHERE product_code IN (
    SELECT pc FROM (SELECT product_code AS pc
        			FROM products
        			GROUP BY product_code
        			HAVING COUNT(*) > 1
    ) AS t
);
